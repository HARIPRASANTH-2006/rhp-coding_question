import java.util.*;

public class Main {

    static int[] parent;

    static int find(int x) {
        if (parent[x] == x) {
            return x;
        }

        return parent[x] = find(parent[x]);
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int T = sc.nextInt();

        for (int tc = 1; tc <= T; tc++) {

            int n = sc.nextInt();
            int q = sc.nextInt();

            int[] A = new int[n];

            for (int i = 0; i < n; i++) {
                A[i] = sc.nextInt();
            }

            parent = new int[100001];

            for (int i = 1; i <= 100000; i++) {
                parent[i] = i;
            }

            System.out.println("Case " + tc + ":");

            while (q-- > 0) {

                int type = sc.nextInt();

                if (type == 1) {

                    int x = sc.nextInt();
                    int y = sc.nextInt();

                    x = find(x);
                    y = find(y);

                    if (x != y) {
                        parent[x] = y;
                    }

                } else {

                    int idx = sc.nextInt();

                    System.out.println(find(A[idx - 1]));
                }
            }
        }

        sc.close();
    }
}
