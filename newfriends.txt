import java.util.*;

public class Main {
    public static int find(int[] arr, int n) {
        if (arr[n] != n) {
            arr[n] = find(arr, arr[n]);
        }
        return arr[n];
    }
    public static void join(int[] arr, int x, int y) {
        int left = find(arr, x);
        int right = find(arr, y);
        arr[right] = left;
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();

        int[] leader = new int[n + 1];
        for (int i=1; i<=n; i++) {
            leader[i] = i;
        }

        int[][] edges = new int[m][2];

        for (int i=0; i<m; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            edges[i][0] = x;
            edges[i][1] = y;
            join(leader, x, y);
        }

        Set<Integer> set = new HashSet<>();

        for (int i=1; i<=n; i++) {
            set.add(find(leader, i));
        }

        int[] grpCnt = new int[n + 1];
        for (int i=1; i<=n; i++) {
            int val = find(leader, i);
            grpCnt[val]++;
        }

        int[] edgeCnt = new int[n + 1];
        for (int i=0; i<m; i++) {
            int u = edges[i][0];
            int v = edges[i][1];
            int val = find(leader, u);
            edgeCnt[val]++;
        }

        long sum = 0;
        for (int i=1; i<=n; i++) {
            if (grpCnt[i] != 0) {
                int x = grpCnt[i];
                int y = edgeCnt[i];
                sum = (long) sum + (((long) x * (x-1)) / 2) - y;
            }
        }

        System.out.println(sum);
    }
}
