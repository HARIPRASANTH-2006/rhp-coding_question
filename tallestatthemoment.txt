import java.util.*;

public class Main {

    private static int upperBound(List<Integer> time,int target) {
        int low = 0;
        int high = time.size() - 1;

        int bound = time.size();

        while(low <= high) {
            int mid = low + (high-low)/2;

            if(time.get(mid) > target) {
                bound = mid;
                high = mid-1;
            }

            else {
                low = mid+1;
            }
        }

        return bound;
    }

    public static int maximumHeight(List<Integer> height, List<Integer> time, int currentTime) {
        int bound = upperBound(time,currentTime);

        return height.get(bound);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int N = sc.nextInt();

        int[][] takashi = new int[N][2];

        for(int i = 0; i < N; i++ ) {
            takashi[i][0] = sc.nextInt(); //height
            takashi[i][1] = sc.nextInt(); //time
        }

        Arrays.sort(takashi, (a, b) -> b[0] - a[0]);

        List<Integer> height = new ArrayList<>();
        List<Integer> time = new ArrayList<>();

        int kill = 0;

        for(int i=0; i < N; i++) {

            if(takashi[i][1] > kill) {

                kill = takashi[i][1];
                height.add(takashi[i][0]);
                time.add(takashi[i][1]);
            }
        }

        int Q = sc.nextInt();
        for(int i = 0; i < Q; i++) {
            int t = sc.nextInt();
            System.out.println(maximumHeight(height,time,t));
        }

        sc.close();
    }

}
