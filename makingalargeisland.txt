import java.util.*;

class Solution {

    int n;
    int[][] grid;

    int[] dr = {-1, 1, 0, 0};
    int[] dc = {0, 0, -1, 1};

    public int largestIsland(int[][] grid) {

        this.grid = grid;
        n = grid.length;

        
        HashMap<Integer, Integer> area = new HashMap<>();

        int id = 2;

       
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {

                if (grid[i][j] == 1) {

                    int size = dfs(i, j, id);

                    area.put(id, size);

                    id++;
                }
            }
        }

        int ans = 0;

        
        for (int value : area.values()) {
            ans = Math.max(ans, value);
        }

        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {

                if (grid[i][j] == 0) {

                    HashSet<Integer> seen = new HashSet<>();

                    int size = 1; 

                    for (int k = 0; k < 4; k++) {

                        int ni = i + dr[k];
                        int nj = j + dc[k];

                        if (ni >= 0 && ni < n &&
                            nj >= 0 && nj < n) {

                            int islandId = grid[ni][nj];

                           
                            if (islandId > 1 &&
                                !seen.contains(islandId)) {

                                size += area.get(islandId);

                                seen.add(islandId);
                            }
                        }
                    }

                    ans = Math.max(ans, size);
                }
            }
        }

        return ans;
    }

    int dfs(int r, int c, int id) {

        if (r < 0 || r >= n ||
            c < 0 || c >= n ||
            grid[r][c] != 1) {

            return 0;
        }

        
        grid[r][c] = id;

        int size = 1;

        for (int k = 0; k < 4; k++) {

            int nr = r + dr[k];
            int nc = c + dc[k];

            size += dfs(nr, nc, id);
        }

        return size;
    }
}
